artifactory-airlift chart installed.

This chart only ships supporting resources (ConfigMap, Secret, PVCs).
The sidecar container itself is added to the artifactory pod via the
jfrog/artifactory chart's `customSidecarContainers` field. Paste the
following block into the jfrog/artifactory Helm values for this
namespace (under `artifactory.artifactory`):

  artifactory:
    artifactory:
      customVolumes: |
        - name: airlift-state
          persistentVolumeClaim:
            claimName: {{ include "airlift.statePvcName" . }}
        - name: airlift-spool
          persistentVolumeClaim:
            claimName: {{ include "airlift.spoolPvcName" . }}
        - name: airlift-config
          configMap:
            name: {{ include "airlift.configMapName" . }}
{{- if .Values.artifactory.caCert.existingConfigMap }}
        # CA bundle for trusting the Artifactory TLS cert. Reference an
        # existing ConfigMap (e.g. kube-root-ca.crt, an injected
        # trusted-CA-bundle, or a trust-manager Bundle).
        - name: airlift-ca
          configMap:
            name: {{ .Values.artifactory.caCert.existingConfigMap | quote }}
{{- end }}
{{- if .Values.artifactory.tokenFile.existingSecret }}
        # Bearer token read from disk on every request, so an externally
        # rotated token is picked up without restarting the pod. Keep this a
        # directory mount: subPath mounts never receive updates.
        - name: airlift-token
          secret:
            secretName: {{ .Values.artifactory.tokenFile.existingSecret | quote }}
{{- end }}
{{- if .Values.binarystore.configFrom.existingSecret }}
        # binarystore.xml read from a Secret rather than Artifactory's own
        # on-disk copy, whose credentials Artifactory encrypts in place. Only
        # the one key is projected: such Secrets often also hold the instance
        # master and join keys.
        - name: airlift-binarystore-config
          secret:
            secretName: {{ .Values.binarystore.configFrom.existingSecret | quote }}
            items:
              - key: {{ .Values.binarystore.configFrom.key | quote }}
                path: {{ .Values.binarystore.configFrom.key | quote }}
{{- end }}
{{- if .Values.binarystore.existingSecret }}
        # Object-storage credentials, for when binarystore.xml cannot supply
        # them (Artifactory rewrites them encrypted in its own copy).
        - name: airlift-binarystore
          secret:
            secretName: {{ .Values.binarystore.existingSecret | quote }}
{{- end }}
{{- if .Values.binarystore.caCert.existingConfigMap }}
        # CA bundle for the object-storage endpoint (may differ from the CA
        # fronting Artifactory itself).
        - name: airlift-binarystore-ca
          configMap:
            name: {{ .Values.binarystore.caCert.existingConfigMap | quote }}
{{- end }}
      customVolumeMounts: |
        - name: airlift-state
          mountPath: {{ .Values.paths.stateMount }}
        - name: airlift-spool
          mountPath: {{ .Values.paths.spoolMount }}
        - name: airlift-config
          mountPath: {{ .Values.paths.configMount }}
      customSidecarContainers: |
        - name: airlift
          image: {{ .Values.image.repository }}:{{ .Values.image.tag }}
          imagePullPolicy: {{ .Values.image.pullPolicy }}
          env:
            - name: AIRLIFT_MODE
              value: {{ .Values.mode | quote }}
{{- if .Values.artifactory.tokenFile.existingSecret }}
            - name: AIRLIFT_ARTIFACTORY_TOKEN_FILE
              value: {{ printf "%s/%s" .Values.artifactory.tokenFile.mountPath .Values.artifactory.tokenFile.key | quote }}
{{- else }}
            - name: AIRLIFT_ARTIFACTORY_TOKEN
              valueFrom:
                secretKeyRef:
                  name: {{ include "airlift.secretName" . }}
                  key: token
            - name: AIRLIFT_ARTIFACTORY_USERNAME
              valueFrom:
                secretKeyRef:
                  name: {{ include "airlift.secretName" . }}
                  key: username
            - name: AIRLIFT_ARTIFACTORY_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: {{ include "airlift.secretName" . }}
                  key: password
{{- end }}
{{- if .Values.artifactory.caCert.existingConfigMap }}
            - name: AIRLIFT_ARTIFACTORY_CA_CERT
              value: {{ printf "%s/%s" .Values.artifactory.caCert.mountPath .Values.artifactory.caCert.key | quote }}
{{- end }}
{{- if .Values.binarystore.configFrom.existingSecret }}
            - name: AIRLIFT_BINARYSTORE_CONFIG
              value: {{ printf "%s/%s" .Values.binarystore.configFrom.mountPath .Values.binarystore.configFrom.key | quote }}
{{- end }}
{{- if .Values.binarystore.existingSecret }}
            - name: AIRLIFT_BINARYSTORE_ACCESS_KEY
              valueFrom:
                secretKeyRef:
                  name: {{ .Values.binarystore.existingSecret | quote }}
                  key: accessKey
                  optional: true
            - name: AIRLIFT_BINARYSTORE_SECRET_KEY
              valueFrom:
                secretKeyRef:
                  name: {{ .Values.binarystore.existingSecret | quote }}
                  key: secretKey
                  optional: true
            - name: AIRLIFT_BINARYSTORE_ACCOUNT_KEY
              valueFrom:
                secretKeyRef:
                  name: {{ .Values.binarystore.existingSecret | quote }}
                  key: accountKey
                  optional: true
{{- end }}
          volumeMounts:
            - name: airlift-state
              mountPath: {{ .Values.paths.stateMount }}
            - name: airlift-spool
              mountPath: {{ .Values.paths.spoolMount }}
            - name: airlift-config
              mountPath: {{ .Values.paths.configMount }}
            - name: artifactory-volume
              mountPath: /var/opt/jfrog/artifactory
{{- if .Values.artifactory.caCert.existingConfigMap }}
            - name: airlift-ca
              mountPath: {{ .Values.artifactory.caCert.mountPath }}
              readOnly: true
{{- end }}
{{- if .Values.artifactory.tokenFile.existingSecret }}
            - name: airlift-token
              mountPath: {{ .Values.artifactory.tokenFile.mountPath }}
              readOnly: true
{{- end }}
{{- if .Values.binarystore.configFrom.existingSecret }}
            - name: airlift-binarystore-config
              mountPath: {{ .Values.binarystore.configFrom.mountPath }}
              readOnly: true
{{- end }}
{{- if .Values.binarystore.existingSecret }}
            - name: airlift-binarystore
              mountPath: {{ .Values.binarystore.mountPath }}
              readOnly: true
{{- end }}
{{- if .Values.binarystore.caCert.existingConfigMap }}
            - name: airlift-binarystore-ca
              mountPath: {{ .Values.binarystore.caCert.mountPath }}
              readOnly: true
{{- end }}
{{- with .Values.sidecar.securityContext }}
          securityContext:
            {{- toYaml . | nindent 12 }}
{{- end }}

Install this chart twice (once per Artifactory instance) with
`mode: sender` on the source side and `mode: receiver` on the
destination side.
